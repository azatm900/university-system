// THEME TOGGLE

(function () {
    const root = document.documentElement;
    const btn = document.getElementById('theme-toggle');
    function apply(theme) {
        if (theme === 'dark') root.classList.add('dark');
        else root.classList.remove('dark');
        try { localStorage.setItem('theme', theme); } catch (e) {}
        if (!btn) return;
        const sun = btn.querySelector('[data-icon="sun"]');
        const moon = btn.querySelector('[data-icon="moon"]');
        if (sun && moon) {
            sun.classList.toggle('hidden', theme === 'dark');
            sun.classList.toggle('flex', theme !== 'dark');
            moon.classList.toggle('hidden', theme !== 'dark');
            moon.classList.toggle('flex', theme === 'dark');
        }
    }
    apply(localStorage.getItem('theme') || 'light');
    if (btn) {
        btn.addEventListener('click', () => {
            apply(root.classList.contains('dark') ? 'light' : 'dark');
        });
    }
})();

// ============================================================
// SIDEBAR (collapse desktop + drawer mobile + body scroll lock)
// ============================================================
(function () {
    const sidebar = document.getElementById('sidebar');
    const btn = document.getElementById('sidebar-toggle');
    const backdrop = document.getElementById('backdrop');
    if (!sidebar || !btn) return;

    function isMobile() { return window.matchMedia('(max-width: 1023px)').matches; }

    // restore collapsed state on desktop
    if (!isMobile() && localStorage.getItem('sidebar-collapsed') === '1') {
        sidebar.classList.add('collapsed');
    }

    function openDrawer() {
        sidebar.classList.add('mobile-open');
        if (backdrop) backdrop.classList.add('show');
        document.body.classList.add('sidebar-open');
    }
    function closeDrawer() {
        sidebar.classList.remove('mobile-open');
        if (backdrop) backdrop.classList.remove('show');
        document.body.classList.remove('sidebar-open');
    }

    btn.addEventListener('click', () => {
        if (isMobile()) {
            if (sidebar.classList.contains('mobile-open')) closeDrawer();
            else openDrawer();
        } else {
            sidebar.classList.toggle('collapsed');
            try {
                localStorage.setItem('sidebar-collapsed', sidebar.classList.contains('collapsed') ? '1' : '0');
            } catch (e) {}
        }
    });

    if (backdrop) backdrop.addEventListener('click', closeDrawer);

    // Close drawer automatically when clicking any nav link (mobile)
    sidebar.querySelectorAll('a[href]').forEach(link => {
        link.addEventListener('click', () => {
            if (isMobile()) closeDrawer();
        });
    });

    // Close drawer on Escape
    document.addEventListener('keydown', (e) => {
        if (e.key === 'Escape' && sidebar.classList.contains('mobile-open')) closeDrawer();
    });

    // If window grows to desktop, ensure mobile state cleaned up
    window.addEventListener('resize', () => {
        if (!isMobile()) {
            sidebar.classList.remove('mobile-open');
            if (backdrop) backdrop.classList.remove('show');
            document.body.classList.remove('sidebar-open');
        }
    });
})();

(function () {
    const badge = document.getElementById('notif-badge');
    const list = document.querySelector('#notif-menu [data-notif-list]');
    if (!badge && !list) return;
    fetch('/api/notifications', {
        headers: { 'Accept': 'application/json', 'X-Requested-With': 'XMLHttpRequest' },
        credentials: 'same-origin',
    })
        .then(r => r.ok ? r.json() : null)
        .then(data => {

                    '<span style="flex:1">' + i.text + '</span>' +
                    (i.count ? '<span class="badge badge-primary">' + i.count + '</span>' : '') +
                    '</a>'
                ).join('');
            }
        })
        .catch(() => {});
})();
