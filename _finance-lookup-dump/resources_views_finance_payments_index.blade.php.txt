<script>
function printA4() {
    document.body.classList.remove('thermal-mode');
    document.getElementById('thermal-page-style').textContent = '';
    setTimeout(() => window.print(), 100);
}

function printThermal() {
    document.body.classList.add('thermal-mode');
    document.getElementById('thermal-page-style').textContent = '@page { size: 80mm auto; margin: 2mm; }';
    setTimeout(() => window.print(), 100);
}

function savePdf() {
    // يفتح print dialog → اختر "Save as PDF"
    document.body.classList.remove('thermal-mode');
    document.getElementById('thermal-page-style').textContent = '@page { size: A4 landscape; margin: 0; }';
    setTimeout(() => window.print(), 100);
}
</script>
